// 傾きによるボール移動と「穴(R)」の判定ロジック例 function updateGame() { if (!gameActive) return; // 1. 傾きでボールの速度を更新 ball.vx += tiltX * 0.05; ball.vy += tiltY * 0.05; ball.vx *= 0.98; // 摩擦 ball.vy *= 0.98; // 2. 移動後の仮座標 let nextX = ball.x + ball.vx; let nextY = ball.y + ball.vy; // 3. 現在地（グリッド座標）の計算 const col = Math.floor(nextX / tileSize); const row = Math.floor(nextY / tileSize); const cell = mazeMap[row] ? mazeMap[row][col] : 1; // 4. 判定処理 if (cell === 1) { // 壁に当たったら跳ね返る ball.vx *= -0.5; ball.vy *= -0.5; } else if (cell === 'R') { // 穴に落ちた！ statusText.innerText = 'あ！穴に落ちちゃった…！'; resetBallToStart(); // スタート位置に戻す } else if (cell === 'G') { // ゴール！ statusText.innerText = 'ゴール達成！おめでとう！'; gameActive = false; } else { // 通路なら移動確定 ball.x = nextX; ball.y = nextY; } }
